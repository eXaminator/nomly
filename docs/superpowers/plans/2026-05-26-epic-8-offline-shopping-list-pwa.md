# Epic 8 — Offline shopping list (PWA) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn `/einkauf` into an installable PWA whose shopping list (view + check / uncheck + supermarket filter) keeps working offline, with a mascot-based icon and home-screen shortcuts.

**Architecture:** Web manifest (Next.js `app/manifest.ts`) + mascot icons rendered from SVGs via `sharp` + hand-rolled service worker (`public/sw.js`) + IndexedDB (via `idb`) mirror of the shopping list and a FIFO mutation queue + connectivity-aware optimistic-update flow on the existing `/einkauf` page. Mutations go over the existing HTTP `PATCH /api/v1/shopping/items/:id` endpoint, extended with an idempotency `mutationId` field. Realtime peer sync (Epic 9) reuses this transport.

**Tech stack:** Next.js 16 (App Router), React 19, TypeScript, Tailwind v4, `idb` (added), `sharp` (existing), `fake-indexeddb` (added dev), Playwright (existing), Vitest (existing).

**Spec reference:** `docs/superpowers/specs/2026-05-26-epic-8-pwa-offline-design.md`. Reference SVG sketches: `docs/superpowers/specs/2026-05-26-epic-8-assets/mascot-reference.svg` and `mascot-maskable-reference.svg`.

**Important — Next.js 16 conventions:** Before writing the manifest module, the icon registration, or theme-color metas, read `node_modules/next/dist/docs/` for the current canonical conventions. If anything below collides with the framework's documented Next.js 16 form, prefer the framework form and adjust the task.

Write `src/lib/shopping/offline/mutation-dedup.ts`:

```ts
const CAPACITY = 256;
const TTL_MS = 10 * 60 * 1000;

type Entry = { value: unknown; recordedAt: number };

const cache = new Map<string, Entry>();

export function recordMutation(
  mutationId: string,
  value: unknown,
  recordedAt: number = Date.now(),
): void {
  if (cache.has(mutationId)) cache.delete(mutationId);
  cache.set(mutationId, { value, recordedAt });
  while (cache.size > CAPACITY) {
    const oldestKey = cache.keys().next().value;
    if (oldestKey === undefined) break;
    cache.delete(oldestKey);
  }
}

export function recallMutation<T = unknown>(mutationId: string): T | null {
  const entry = cache.get(mutationId);
  if (!entry) return null;
  if (Date.now() - entry.recordedAt > TTL_MS) {
    cache.delete(mutationId);
    return null;
  }
  // Refresh insertion order so frequent reuses behave LRU-ish.
  cache.delete(mutationId);
  cache.set(mutationId, entry);
  return entry.value as T;
}

export function __resetMutationCacheForTests(): void {
  cache.clear();
}
```

```ts
import { NextResponse } from "next/server";
import { z } from "zod";
import { deleteItem, patchItem, ShoppingApiError } from "@/lib/shopping/items";
import {
  recallMutation,
  recordMutation,
} from "@/lib/shopping/offline/mutation-dedup";

const patchSchema = z
  .object({
    checked: z.boolean().optional(),
    note: z.string().nullable().optional(),
    mutationId: z.string().uuid().optional(),
  })
  .strict();

function parseId(raw: string): number | null {
  const n = Number(raw);
  return Number.isInteger(n) && n > 0 ? n : null;
}

export async function PATCH(
  request: Request,
  ctx: { params: Promise<{ id: string }> },
) {
  const { id: rawId } = await ctx.params;
  const id = parseId(rawId);
  if (!id) return NextResponse.json({ error: "not found" }, { status: 404 });
  const body = await request.json();
  const parsed = patchSchema.safeParse(body);
  if (!parsed.success) {
    return NextResponse.json(
      { error: "validation", details: parsed.error.flatten() },
      { status: 400 },
    );
  }
  const { mutationId, ...patch } = parsed.data;
  if (mutationId) {
    const cached = recallMutation<{ item: unknown }>(mutationId);
    if (cached) return NextResponse.json(cached);
  }
  try {
    const item = patchItem(id, patch);
    const payload = { item };
    if (mutationId) recordMutation(mutationId, payload);
    return NextResponse.json(payload);
  } catch (err) {
    if (err instanceof ShoppingApiError) {
      return NextResponse.json({ error: err.message }, { status: err.status });
    }
    throw err;
  }
}

export async function DELETE(
  _request: Request,
  ctx: { params: Promise<{ id: string }> },
) {
  const { id: rawId } = await ctx.params;
  const id = parseId(rawId);
  if (!id) return NextResponse.json({ error: "not found" }, { status: 404 });
  try {
    deleteItem(id);
    return NextResponse.json({ ok: true });
  } catch (err) {
    if (err instanceof ShoppingApiError) {
      return NextResponse.json({ error: err.message }, { status: err.status });
    }
    throw err;
  }
}
```

```ts
import sharp from "sharp";
import { readFile, writeFile } from "node:fs/promises";
import { join } from "node:path";
import { fileURLToPath } from "node:url";
import { dirname } from "node:path";

const ICON_SIZES = [180, 192, 384, 512] as const;
const SHORTCUT_KEYS = ["einkauf", "plan", "rezepte", "neu"] as const;

async function renderTo(svgPath: string, outPath: string, size: number): Promise<void> {
  const buf = await readFile(svgPath);
  const png = await sharp(buf, { density: 384 }).resize(size, size).png().toBuffer();
  await writeFile(outPath, png);
}

export async function renderIcons(dir: string): Promise<void> {
  const iconSvg = join(dir, "icon.svg");
  for (const size of ICON_SIZES) {
    await renderTo(iconSvg, join(dir, `icon-${size}.png`), size);
  }
  const maskableSvg = join(dir, "icon-maskable.svg");
  await renderTo(maskableSvg, join(dir, "icon-maskable-512.png"), 512);

  // Shortcut icons are optional in the test harness — only render if the
  // source exists.
  for (const key of SHORTCUT_KEYS) {
    const src = join(dir, `shortcut-${key}.svg`);
    try {
      await readFile(src);
      await renderTo(src, join(dir, `shortcut-${key}-96.png`), 96);
    } catch {
      // missing — skip
    }
  }
}

// Allow running as a CLI: `node --experimental-transform-types ... scripts/render-icons.mts`
const isMain = process.argv[1] === fileURLToPath(import.meta.url);
if (isMain) {
  const here = dirname(fileURLToPath(import.meta.url));
  const target = join(here, "..", "public", "icons");
  await renderIcons(target);
  console.log(`> rendered icons into ${target}`);
}
```

```ts
type BuildManifest = {
  pages?: Record<string, string[]>;
  rootMainFiles?: string[];
};

async function loadBuildManifest(): Promise<BuildManifest | null> {
  for (const candidate of [".next/app-build-manifest.json", ".next/build-manifest.json"]) {
    try {
      const raw = await readFile(candidate, "utf8");
      return JSON.parse(raw) as BuildManifest;
    } catch {
      continue;
    }
  }
  return null;
}

async function main(): Promise<void> {
  const manifest = await loadBuildManifest();
  const assets = new Set<string>();

  if (manifest?.pages) {
    for (const [page, files] of Object.entries(manifest.pages)) {
      if (page === "/einkauf" || page === "/_app" || page === "/_layout") {
        for (const f of files) assets.add(`/_next/${f}`);
      }
    }
  }
  if (manifest?.rootMainFiles) {
    for (const f of manifest.rootMainFiles) assets.add(`/_next/${f}`);
  }

  // App shell extras the SW always wants cached on install.
  const shell = [
    "/einkauf",
    "/manifest.webmanifest",
    "/icons/icon.svg",
    "/icons/icon-192.png",
  ];

  const payload = { shell, assets: Array.from(assets).sort() };
  await writeFile(join("public", "sw-precache.json"), JSON.stringify(payload, null, 2));
  console.log(`> wrote public/sw-precache.json (${shell.length} shell + ${assets.size} assets)`);
```

```js
/* eslint-disable */
// nomly service worker — Epic 8.
// Caches the /einkauf shell + static chunks (cache-first) and the shopping
// + supermarket + category GET endpoints (network-first with cache fallback).
// Never intercepts non-GET — the offline mutation queue lives client-side.

const CACHE_VERSION = "v1";
const SHELL_CACHE  = `nomly-shell-${CACHE_VERSION}`;
const DATA_CACHE   = `nomly-data-${CACHE_VERSION}`;
const STATIC_CACHE = `nomly-static-${CACHE_VERSION}`;
const ALL_CACHES   = new Set([SHELL_CACHE, DATA_CACHE, STATIC_CACHE]);

self.addEventListener("install", (event) => {
  event.waitUntil((async () => {
    try {
      const res = await fetch("/sw-precache.json", { cache: "no-store" });
      if (res.ok) {
        const { shell = [], assets = [] } = await res.json();
        const cache = await caches.open(SHELL_CACHE);
        await cache.addAll([...shell, ...assets].filter(Boolean));
      }
    } catch (e) {
      // Precache failure is non-fatal — the SW still installs; missing
      // assets just stream from the network until they're cached on demand.
      console.warn("[sw] precache failed", e);
    }
    await self.skipWaiting();
  })());
});

self.addEventListener("activate", (event) => {
  event.waitUntil((async () => {
    const names = await caches.keys();
    await Promise.all(
      names
        .filter((n) => n.startsWith("nomly-") && !ALL_CACHES.has(n))
        .map((n) => caches.delete(n)),
    );
    await self.clients.claim();
  })());
});

self.addEventListener("message", (event) => {
  if (event.data === "SKIP_WAITING") self.skipWaiting();
});

const DATA_PATHS = ["/api/v1/shopping", "/api/v1/supermarkets", "/api/v1/categories"];
function isDataPath(url) {
  return DATA_PATHS.some((p) => url.pathname === p || url.pathname.startsWith(p + "/"));
}
function isStaticAsset(url) {
  return url.pathname.startsWith("/_next/static/") || url.pathname.startsWith("/icons/");
}
function isShellNavigation(req, url) {
  return req.mode === "navigate" && url.pathname === "/einkauf";
}

self.addEventListener("fetch", (event) => {
  const req = event.request;
  if (req.method !== "GET") return;
  const url = new URL(req.url);
  if (url.origin !== self.location.origin) return;

  if (isShellNavigation(req, url)) {
    event.respondWith(shellNavigationStrategy(req));
    return;
  }
  if (isDataPath(url)) {
    event.respondWith(networkFirst(req, DATA_CACHE));
    return;
  }
  if (isStaticAsset(url)) {
    event.respondWith(cacheFirst(req, STATIC_CACHE));
    return;
  }
  // Everything else: pass through.
});

async function shellNavigationStrategy(req) {
  const cache = await caches.open(SHELL_CACHE);
  const cached = await cache.match("/einkauf");
  // Kick the refresh in the background so the cache stays current.
  fetch(req).then((res) => {
    if (res.ok) cache.put("/einkauf", res.clone());
  }).catch(() => undefined);
  if (cached) return cached;
  try {
    const fresh = await fetch(req);
    if (fresh.ok) cache.put("/einkauf", fresh.clone());
    return fresh;
  } catch {
    return new Response(
      "<!doctype html><meta charset=utf-8><title>nomly offline</title><p>Offline — Einkaufsliste konnte nicht geladen werden. Öffne die Seite einmal online, dann steht sie auch offline bereit.</p>",
      { headers: { "content-type": "text/html;charset=utf-8" }, status: 503 },
    );
  }
}

async function networkFirst(req, cacheName) {
  const cache = await caches.open(cacheName);
  try {
    const res = await fetch(req);
    if (res.ok) cache.put(req, res.clone());
    return res;
  } catch (err) {
    const cached = await cache.match(req);
    if (cached) return cached;
    throw err;
  }
}

async function cacheFirst(req, cacheName) {
  const cache = await caches.open(cacheName);
  const cached = await cache.match(req);
  if (cached) return cached;
  const res = await fetch(req);
  if (res.ok) cache.put(req, res.clone());
  return res;
}
```

```ts
export type ConnectivityState =
  | "online"
  | "offline"
  | "syncing"
  | "sessionExpired";

type Listener = (state: ConnectivityState) => void;

let state: ConnectivityState = "online";
const listeners = new Set<Listener>();

export function getConnectivity(): ConnectivityState {
  return state;
}

export function setConnectivity(next: ConnectivityState): void {
  if (next === state) return;
  state = next;
  for (const l of listeners) l(state);
}

export function subscribeConnectivity(l: Listener): () => void {
  listeners.add(l);
  return () => {
    listeners.delete(l);
  };
}

const HEARTBEAT_MS = 30_000;
let heartbeatHandle: ReturnType<typeof setInterval> | null = null;

async function pingHealth(): Promise<boolean> {
  try {
    const res = await fetch("/api/health", {
      method: "HEAD",
      cache: "no-store",
      signal: AbortSignal.timeout(5_000),
    });
    return res.ok;
  } catch {
    return false;
  }
}

export function startConnectivityWatcher(): () => void {
  if (typeof window === "undefined") return () => undefined;

  const onOnline = () => setConnectivity("online");
  const onOffline = () => setConnectivity("offline");
  const onVisibility = async () => {
    if (document.visibilityState !== "visible") return;
    const ok = await pingHealth();
    setConnectivity(ok ? "online" : "offline");
  };

  window.addEventListener("online", onOnline);
  window.addEventListener("offline", onOffline);
  document.addEventListener("visibilitychange", onVisibility);

  // Seed the initial state.
  if (navigator.onLine === false) setConnectivity("offline");

  heartbeatHandle = setInterval(async () => {
    if (document.visibilityState !== "visible") return;
    const ok = await pingHealth();
    if (state === "sessionExpired") return; // sticky until user re-auths
    setConnectivity(ok ? "online" : "offline");
  }, HEARTBEAT_MS);

  return () => {
    window.removeEventListener("online", onOnline);
    window.removeEventListener("offline", onOffline);
    document.removeEventListener("visibilitychange", onVisibility);
    if (heartbeatHandle) clearInterval(heartbeatHandle);
    heartbeatHandle = null;
  };
}
```

```ts
import { openOfflineDb } from "./db";
import type { ShoppingItemView } from "@/lib/shopping/items";

// Replace the IDB items store with the server snapshot, then overlay any
// locally-pending toggles so the user's optimistic state survives a
// re-fetch. Returns the merged list.
export async function hydrateFromIDB(
  serverItems: ShoppingItemView[],
): Promise<ShoppingItemView[]> {
  const db = await openOfflineDb();

  // Stage 1: snapshot replace.
  const tx = db.transaction(["items", "meta"], "readwrite");
  await tx.objectStore("items").clear();
  for (const item of serverItems) {
    await tx.objectStore("items").put(item);
  }
  await tx.objectStore("meta").put({ key: "shoppingFetchedAt", value: Date.now() });
  await tx.done;

  // Stage 2: overlay pending toggles.
  const queueEntries = await db.getAll("mutationQueue");
  if (queueEntries.length === 0) return serverItems;

  const byId = new Map(serverItems.map((i) => [i.id, i] as const));
  for (const entry of queueEntries) {
    const base = byId.get(entry.itemId);
    if (!base) continue;
    byId.set(entry.itemId, {
      ...base,
      checked: entry.type === "check",
      checkedAt: entry.type === "check" ? Math.floor(entry.queuedAt / 1000) : null,
    });
  }
  return Array.from(byId.values());
}

export async function toggleItemOptimistic(
  itemId: number,
  nextChecked: boolean,
): Promise<void> {
  const db = await openOfflineDb();
  const mutationId = crypto.randomUUID();
  const now = Date.now();

  const tx = db.transaction(["items", "mutationQueue"], "readwrite");
  const current = await tx.objectStore("items").get(itemId);
  if (current) {
    await tx.objectStore("items").put({
      ...current,
      checked: nextChecked,
      checkedAt: nextChecked ? Math.floor(now / 1000) : null,
    });
  }
  await tx.objectStore("mutationQueue").put({
    mutationId,
    itemId,
    type: nextChecked ? "check" : "uncheck",
    queuedAt: now,
    attempts: 0,
  });
  await tx.done;
}

export async function readItemsFromIDB(): Promise<ShoppingItemView[]> {
  const db = await openOfflineDb();
  return db.getAll("items");
}

export async function readPendingQueue() {
  const db = await openOfflineDb();
  const tx = db.transaction("mutationQueue", "readonly");
  return tx.store.index("byQueuedAt").getAll();
}
```

```ts
import { openOfflineDb } from "./db";
import { patchItem as apiPatchItem } from "@/components/shopping/api";
import { setConnectivity } from "./connectivity";

export type DrainResult = {
  drained: number;
  remaining: number;
  lastErrorKind?: "network" | "auth" | "server";
};

function isAuthErrorMessage(message: string): boolean {
  // Matches the shape produced by `jsonOrThrow` in components/shopping/api.ts.
  return /\(401\)/.test(message);
}

export async function drainQueueOnce(): Promise<DrainResult> {
  const db = await openOfflineDb();
  let drained = 0;
  let lastErrorKind: DrainResult["lastErrorKind"] | undefined;

  while (true) {
    const tx = db.transaction("mutationQueue", "readonly");
    const ordered = await tx.store.index("byQueuedAt").getAll();
    if (ordered.length === 0) break;
    const entry = ordered[0];

    try {
      const updated = await apiPatchItem(entry.itemId, {
        checked: entry.type === "check",
        mutationId: entry.mutationId,
      });
      const writeTx = db.transaction(["items", "mutationQueue"], "readwrite");
      await writeTx.objectStore("items").put(updated);
      await writeTx.objectStore("mutationQueue").delete(entry.mutationId);
      await writeTx.done;
      drained++;
    } catch (err) {
      const message = err instanceof Error ? err.message : String(err);
      const kind: DrainResult["lastErrorKind"] = isAuthErrorMessage(message)
        ? "auth"
        : err instanceof TypeError
          ? "network"
          : "server";
      await db.put("mutationQueue", {
        ...entry,
        attempts: entry.attempts + 1,
        lastError: message,
      });
      lastErrorKind = kind;
      if (kind === "auth") setConnectivity("sessionExpired");
      else if (kind === "network") setConnectivity("offline");
      break;
    }
  }

  const tx = db.transaction("mutationQueue", "readonly");
  const remaining = (await tx.store.getAll()).length;
  return { drained, remaining, lastErrorKind };
}

let draining = false;
let again = false;

export async function kickSync(): Promise<void> {
  if (draining) {
    again = true;
    return;
  }
  draining = true;
  try {
    do {
      again = false;
      await drainQueueOnce();
    } while (again);
  } finally {
    draining = false;
  }
}
```
